Create src/flights/db-schema.ts with the following Drizzle schema using drizzle-orm/pg-core:

1. priceHistory table ('price_history'):
   - id: serial primary key
   - origin: text not null
   - destination: text not null
   - cabin: text not null ('economy' | 'business' | 'first' | 'premium_economy')
   - program: text not null (scraper key, e.g. 'aa-cdp')
   - flight_number: text nullable
   - departure_date: text not null ('YYYY-MM-DD')
   - departure_time: text nullable
   - miles: integer not null
   - cash_price_usd: numeric(10,2) nullable
   - taxes_usd: numeric(10,2) nullable
   - availability_type: text not null default 'confirmed'
   - scrape_cycle_id: text nullable (ISO timestamp of the cycle, e.g. '2026-02-26T14:30:00.000Z')
   - scraped_at: timestamp defaultNow not null
   Indexes:
   - ph_route_cabin_idx on (origin, destination, cabin, program, departure_date)
   - ph_scraped_at_idx on (scraped_at)

2. alertSubscriptions table ('alert_subscriptions'):
   - id: serial primary key
   - user_id: text not null (Better Auth user ID — foreign key not enforced yet, Phase 6 wires it)
   - origin: text not null
   - destination: text not null
   - cabin: text not null
   - program: text nullable (null = any program)
   - max_miles: integer nullable (null = any price)
   - channel: text not null ('email' | 'whatsapp')
   - contact: text not null (email address or WhatsApp number)
   - active: boolean not null default true
   - created_at: timestamp defaultNow not null
   - updated_at: timestamp defaultNow not null

```typescript
// src/flights/db-drizzle.ts
// Drizzle ORM client for Vercel/Next.js reads.
// The daemon write path continues to use the raw pg pool in db.ts.
import { drizzle } from 'drizzle-orm/node-postgres';
import { Pool } from 'pg';
import * as schema from './db-schema.js';

let _db: ReturnType<typeof drizzle> | null = null;

export function getDrizzle() {
  if (_db) return _db;
  const pool = new Pool({
    connectionString: process.env.DATABASE_URL!,
    ssl: process.env.PG_SSL === 'false' ? false : { rejectUnauthorized: false },
    max: 5, // lower limit for Vercel serverless cold starts
  });
  _db = drizzle(pool, { schema });
  return _db;
}
```

Note: Do NOT import from db-drizzle.ts anywhere in daemon code (src/flights/). It is exclusively for the Next.js app/ directory. This prevents daemon dependencies from bleeding into Vercel builds.
